'use client';

import { useState, useRef } from 'react';

// Interfaz para TypeScript que define la estructura del objeto SOAP
interface SoapData {
  subjetivo?: string;
  objetivo?: string;
  analisis?: string;
  plan?: string;
}

export default function ConsultaPage() {
  // --- ESTADOS ---
  const [grabando, setGrabando] = useState<boolean>(false);
  const [cargando, setCargando] = useState<boolean>(false);
  const [transcripcion, setTranscripcion] = useState<string>('');
  const [notaSoap, setNotaSoap] = useState<SoapData | null>(null);
  const [error, setError] = useState<string | null>(null);

  // --- REFERENCIAS (Para almacenar objetos entre renderizados sin reiniciar el componente) ---
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const audioChunksRef = useRef<Blob[]>([]);


  const formatearValorSOAP = (valor: any): string => {
    if (!valor) return 'Sin datos registrados';
    if (typeof valor === 'object') {
      return Object.entries(valor)
        .map(([clave, val]) => {
          const claveFormateada = clave.replace(/_/g, ' ').replace(/^\w/, (c) => c.toUpperCase());
          return `${claveFormateada}: ${val}`;
        })
        .join('\n');
    }
    return String(valor);
  };

  /**
   * 1. Inicia la grabación de audio usando la API del navegador (MediaRecorder)
   */
  const iniciarGrabacion = async () => {
    setError(null);
    audioChunksRef.current = [];

    try {
      // Solicitar permiso de micrófono
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      const mediaRecorder = new MediaRecorder(stream);
      mediaRecorderRef.current = mediaRecorder;

      // Escuchar eventos para recopilar los fragmentos de audio
      mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          audioChunksRef.current.push(event.data);
        }
      };

      // Iniciar captura
      mediaRecorder.start();
      setGrabando(true);
    } catch (err: any) {
      console.error('Error al acceder al micrófono:', err);
      setError('No se pudo acceder al micrófono. Por favor verifica los permisos.');
    }
  };

  /**
   * 2. Detiene la grabación, empaqueta el audio y lo envía a la API
   */
  const detenerGrabacion = () => {
    if (!mediaRecorderRef.current) return;

    mediaRecorderRef.current.stop();
    setGrabando(false);

    // Al detener la grabadora, procesamos los bloques de audio capturados
    mediaRecorderRef.current.onstop = async () => {
      const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
      
      // Detener todas las pistas de audio para apagar el micrófono en el hardware
      mediaRecorderRef.current?.stream.getTracks().forEach((track) => track.stop());

      // Enviar a la API de Next.js
      await procesarAudio(audioBlob);
    };
  };

  /**
   * 3. Envía el Blob de audio al backend (/api/transcribir) y procesa la respuesta
   */
  const procesarAudio = async (blob: Blob) => {
    setCargando(true);
    setError(null);

    try {
      const formData = new FormData();
      formData.append('audio', blob, 'grabacion.webm');

      // Petición al endpoint backend
      const response = await fetch('/api/transcribir', {
        method: 'POST',
        body: formData,
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.error || 'Error al procesar la nota en el servidor');
      }

      // Guardar datos devueltos en el estado de React
      setTranscripcion(data.data?.transcripcion || '');
      setNotaSoap(data.soap || null);

    } catch (err: any) {
      console.error('Error procesando el audio:', err);
      setError(err.message || 'Ocurrió un error inesperado al procesar la consulta.');
    } finally {
      setCargando(false);
    }
  };

  return (
    <div className="max-w-4xl mx-auto p-6 space-y-8">
      {/* Encabezado */}
      <header className="text-center space-y-2">
        <h1 className="text-3xl font-bold text-gray-900">
          Nueva Consulta Médica (Scribe)
        </h1>
        <p className="text-gray-600">
          Graba la consulta con el paciente para generar automáticamente la nota SOAP con IA.
        </p>
      </header>

      {/* Control de Grabación */}
      <div className="flex flex-col items-center justify-center p-6 bg-white rounded-xl border border-gray-200 shadow-sm">
        {!grabando ? (
          <button
            onClick={iniciarGrabacion}
            disabled={cargando}
            className="flex items-center gap-2 px-6 py-3 bg-red-600 hover:bg-red-700 text-white font-medium rounded-full transition shadow-md disabled:opacity-50"
          >
            <span className="w-3 h-3 bg-white rounded-full animate-pulse"></span>
            Iniciar Grabación
          </button>
        ) : (
          <button
            onClick={detenerGrabacion}
            className="flex items-center gap-2 px-6 py-3 bg-gray-800 hover:bg-gray-900 text-white font-medium rounded-full transition shadow-md"
          >
            <span className="w-3 h-3 bg-red-500 rounded-sm"></span>
            Detener y Generar Nota
          </button>
        )}

        {grabando && (
          <p className="mt-3 text-sm text-red-600 font-medium animate-pulse">
            Grabando audio en vivo...
          </p>
        )}
      </div>

      {/* Indicador de Carga */}
      {cargando && (
        <div className="p-4 bg-blue-50 border border-blue-200 rounded-lg flex items-center justify-center gap-3">
          <div className="animate-spin rounded-full h-5 w-5 border-b-2 border-blue-600"></div>
          <p className="text-blue-700 font-medium">
            Procesando audio y estructurando nota SOAP...
          </p>
        </div>
      )}

      {/* Visualizador de Errores */}
      {error && (
        <div className="p-4 bg-red-50 border border-red-200 text-red-700 rounded-lg text-sm">
          <strong>Error:</strong> {error}
        </div>
      )}

      {/* Renderizado de Resultados (Nota SOAP) */}
      {notaSoap && (
        <section className="bg-white p-6 rounded-xl shadow-sm border border-gray-200 space-y-6">
          <h2 className="text-2xl font-bold text-gray-800 border-b pb-3">
            Nota Médica SOAP Generada
          </h2>

          <div className="grid grid-cols-1 gap-4">
            {/* Subjetivo */}
            <div className="p-4 bg-gray-50 rounded-lg border border-gray-100">
              <h3 className="font-semibold text-blue-700 text-lg mb-1">
                S - Subjetivo
              </h3>
              <p className="text-gray-700 whitespace-pre-wrap leading-relaxed">
                {formatearValorSOAP(notaSoap.objetivo)}
              </p>
            </div>

            {/* Objetivo */}
            <div className="p-4 bg-gray-50 rounded-lg border border-gray-100">
              <h3 className="font-semibold text-blue-700 text-lg mb-1">
                O - Objetivo
              </h3>
             <p className="text-gray-700 whitespace-pre-wrap leading-relaxed">
              {formatearValorSOAP(notaSoap.subjetivo)}
            </p>
            </div>

            {/* Análisis */}
            <div className="p-4 bg-gray-50 rounded-lg border border-gray-100">
              <h3 className="font-semibold text-blue-700 text-lg mb-1">
                A - Evaluación / Análisis
              </h3>
              <p className="text-gray-700 whitespace-pre-wrap leading-relaxed">
                {formatearValorSOAP(notaSoap.analisis)}
              </p>
            </div>

            {/* Plan */}
            <div className="p-4 bg-gray-50 rounded-lg border border-gray-100">
              <h3 className="font-semibold text-blue-700 text-lg mb-1">
                P - Plan
              </h3>
              <p className="text-gray-700 whitespace-pre-wrap leading-relaxed">
                 {formatearValorSOAP(notaSoap.plan)}
              </p>
            </div>
          </div>

          {/* Transcripción completa desplegable */}
          {transcripcion && (
            <details className="mt-4 pt-4 border-t border-gray-100">
              <summary className="font-medium text-gray-600 cursor-pointer hover:text-gray-800">
                Ver transcripción original de voz
              </summary>
              <p className="mt-3 text-sm text-gray-500 bg-gray-50 p-3 rounded border border-gray-100 leading-relaxed">
                {transcripcion}
              </p>
            </details>
          )}
        </section>
      )}
    </div>
  );
}
